// ── CMS-05: konfigurerbare roller (rettigheder som data, ikke hardkodet) ────
model Role {
  id          String   @id @default(cuid())
  navn        String   @unique // fx "Ansvarshavende redaktør"
  permissions Json     // string[] — se PERMISSIONS i lib/permissions.ts
  description String?

  createdAt   DateTime @default(now())
}

model User {
  id           String    @id @default(cuid())
  email        String    @unique
  passwordHash String
  navn         String
  roleId       String
  role         Role      @relation(fields: [roleId], references: [id])
  instansId    String
  instans      Instance  @relation(fields: [instansId], references: [id])
  authorId     String?   @unique // kobling til Forfatter-profil, hvis redaktionel
  author       Author?   @relation(fields: [authorId], references: [id])
  /// Midlertidig adgangskode (ny bruger / nulstilling / seed:prod): al adgang spærres, indtil brugeren har valgt en ny.
  mustChangePassword Boolean @default(false)
  /// Sessioner (JWT) udstedt FØR dette tidspunkt afvises (lib/session-validity.ts) — dræber stjålne sessioner ved kodeskift/nulstilling.
  passwordChangedAt  DateTime?
  /// Deaktiveret bruger: kan ikke logge ind og eksisterende sessioner afvises. Brugeren slettes aldrig (revisioner, opgaver, honorar peger på den).
  deaktiveretTid     DateTime?
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt

  articleRevisions ArticleRevision[]
  createdAssignments Assignment[] @relation("AssignmentCreator")
  approvedHonorEntries HonorEntry[] @relation("HonorApprover")
  /// Ekstra byer (ud over hjemmeinstansen `instansId`) brugeren må redigere — se UserInstanceAccess.
  instanceAccess UserInstanceAccess[]
}

/// Netværksadgang: giver én bruger lov til at arbejde i en ANDEN instans (by) end sin hjemmeinstans (User.instansId).
/// Hjemmeinstansen er altid implicit tilladt og gemmes ikke her. Brugerens ene globale rolle gælder i alle tilladte instanser.
/// Den AKTIVE instans (den by redaktøren arbejder i) ligger i sessionens JWT og valideres mod denne tabel ved hver forespørgsel
/// (lib/instance-access.ts) — slettes en række, falder brugeren tilbage til hjemmeinstansen ved næste forespørgsel.
model UserInstanceAccess {
  id        String   @id @default(cuid())
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  instansId String
  instans   Instance @relation(fields: [instansId], references: [id])
  /// E-mail på den der gav adgangen (eller "cli"); aldrig hemmeligheder.
  createdBy String?
  createdAt DateTime @default(now())

  @@unique([userId, instansId])
  @@index([instansId])
}

/// Revisionsspor for administrative handlinger (brugere, roller, adgangskoder). Aldrig hemmeligheder i `detail`.
/// Bevidst uden relationer: sporet skal overleve, at en bruger/instans ændres, og må aldrig blokere en sletning.
model AuditLog {
  id          String   @id @default(cuid())
  instansId   String
  actorId     String?  // null = driftsværktøj (CLI)
  actorLabel  String?  // e-mail på udførende, som den var dengang
  action      String   // fx "user.create", "user.reset_password"
  targetId    String?
  targetLabel String?  // e-mail på berørt bruger
  detail      Json?    // små, ikke-hemmelige nøgler, fx { fromRole, toRole }
  ip          String?
  createdAt   DateTime @default(now())

  @@index([instansId, createdAt])
}

// ── CMS-01: artikel ─────────────────────────────────────────────────────────
model Article {
  id            String    @id @default(cuid())
  titel         String
  manchet       String?
  slug          String    @unique
  // Blokeditor: body som struktureret komponentliste (jf. spec 4.1), ikke fri HTML.
  blocks        Json      // Block[] — se lib/blocks/schema.ts
  // Status-workflow præcis som spec 5.2 (String + TS-union, valideres i lib/workflow.ts).
  status        String    @default("Idé")
  // indholdstype styrer mærkning (spec 5.2).
  indholdstype  String    @default("Uafhængig") // Uafhængig|Partner|Sponsoreret|Brugerindsendt|PR
  aiBrug        Json      // string[] — obligatorisk AI-brugsfelt ved publicering (spec 3.5)
  // Mærkning for kommercielt indhold (AC-01): skal være udfyldt hvis indholdstype er kommerciel.
  marking       Json?     // { sponsor, aftaleId?, labelTekst, … }
  pinned        Boolean   @default(false)
  breaking      Boolean   @default(false)
  seoTitel      String?
  seoBeskrivelse String?
  sprog         String    @default("da")
  publiceretTid DateTime?
  // Planlagt udgivelse: kun artikler i status "Planlagt" med planlagtTid <= nu flyttes af /api/cron/publish-scheduled.
  planlagtTid   DateTime?
  opdateretTid  DateTime  @updatedAt
  createdAt     DateTime  @default(now())

  kategoriId String?
  kategori   Category? @relation(fields: [kategoriId], references: [id])
  forfatterId String?
  forfatter   Author?   @relation(fields: [forfatterId], references: [id])
  instansId   String
  instans     Instance  @relation(fields: [instansId], references: [id])

  // Mange-til-mange: tags, geotags, relaterede artikler.
  tags       Tag[]
  geoTags    GeoTag[]
  supportAftaleId String?
  supportAftale   SupportAgreement? @relation(fields: [supportAftaleId], references: [id])
  coverMediaId String?
  coverMedia   Media? @relation("ArticleCover", fields: [coverMediaId], references: [id])

  revisions ArticleRevision[]
  assignment Assignment?
  honorEntry HonorEntry?
  corrections Correction[]
  frontpagePlacements FrontpagePlacement[]
  metric ArticleMetric?
  submissions Submission[]
  sourceQAs          SourceQA[]
  interviews         InterviewSession[]
  sponsorBriefs      SponsorBrief[]
  meddelerSager      MeddelerSag[]

  // Agent-indtag (POST /api/ingest/articles): idempotens-nøgle + proveniens.
  // Begge er null for manuelt skrevne artikler (flere NULL er tilladt i unique-indekset).
  externalId    String?
  provenance    Json?     // { ingestKeyPrefix, receivedAt, agent, kilder:[{url,dato,titel,sourceType}], signalIds }

  // Udvidet metadata/SoMe (1:1) og omdirigeringer fra tidligere URL'er.
  meta          ArticleMeta?
  slugRedirects SlugRedirect[]

  @@unique([instansId, externalId])
  @@index([instansId, status])
  @@index([instansId, indholdstype])
  @@index([instansId, status, planlagtTid])
}

/// Udvidet artikelmetadata (SEO, OG/Twitter, opslagstekster pr. platform, Google News/Discover, schema.org).

  id         String   @id @default(cuid())
  articleId  String   @unique
  article    Article  @relation(fields: [articleId], references: [id], onDelete: Cascade)
  instansId  String

  canonicalUrl     String?

  robotsNofollow   Boolean  @default(false)
  keywords         Json     // string[]
  newsKeywords     Json     // string[]
  ogTitel          String?
  ogBeskrivelse    String?
  ogMediaId        String?
  twitterCard      String?  // summary | summary_large_image
  twitterTitel     String?
  twitterBeskrivelse String?
  twitterMediaId   String?
  social           Json     // { facebook|instagram|linkedin|x|bluesky: { tekst, hashtags[], link?, utm? } }
  schemaType       String?  // NewsArticle | OpinionNewsArticle | Article | Report …
  isAccessibleForFree Boolean @default(true)
  paywall          Json?    // { cssSelector }
  dateline         String?
  standout         Boolean  @default(false)
  laesetidMin      Int?
  udloebTid        DateTime?
  begivenhedTid    DateTime?
  medforfattere    Json     // [{ authorId?, navn, rolle }]
  kilder           Json     // [{ titel, url?, udgiver?, dato? }]
  sistSubstantielOpdateringTid DateTime?
  oversaettelser   Json?    // { <sprog>: { url, titel? } }
  createdAt        DateTime @default(now())
  updatedAt        DateTime @updatedAt

  @@index([instansId])
}

/// Tidligere URL (sektion + slug) på en publiceret artikel -> artiklens nuværende URL (301/permanent redirect).

  createdAt    DateTime @default(now())

  @@unique([instansId, fraSektion, fraSlug])
  @@index([tilArticleId])
}

// ── Signaler ──────────────────────────────────────────────────────────────────
model Signal {
  id         String   @id @default(cuid())
  overskrift String
  brødtekst  String?
  kilde      String   @default("Intern") // Ritzau | Reuters | AP | Intern | ...
  kildeUrl   String?
  notable    Boolean  @default(false)
  breaking   Boolean  @default(false)
  laest      Boolean  @default(false)
  version    Int      @default(1)
  instansId  String
  instans    Instance @relation(fields: [instansId], references: [id])
  createdAt  DateTime @default(now())

  // Agent-indtag (POST /api/ingest/signals) — alle felter er valgfrie/additive.
  externalId      String?   // leverandørens stabile id -> idempotent upsert
  kildeUrlNorm    String?   // normaliseret kildeUrl -> dedupe på tværs af externalIds
  sourceType      String?   // kommune_dagsorden | politi | beredskab_112 | ... (valideres i lib/ingest/schema.ts)
  omraadeId       String?
  omraade         GeoTag?   @relation(fields: [omraadeId], references: [id])
  omraadeTekst    String?   // rå geografi-hint hvis ikke matchet til et GeoTag
  kildeTidspunkt  DateTime? // hvornår kilden blev udgivet (publishedAt)
  maskinindsamlet Boolean   @default(false) // vises som "Maskinindsamlet – ikke redaktionelt vurderet"
  ingestKeyId     String?   // hvilken ApiKey leverede signalet
  meta            Json?     // skalare agent-metadata (<= 20 felter); gemmes uændret, vises aldrig offentligt
  // Redaktionel godkendelse (T5 P1-3): maskinsignaler vises KUN offentligt når en redaktør har godkendt dem.
  // Nulstilles når indholdet ændres via indtaget (ny version skal vurderes igen).
  godkendtAf      String?   // User.id på redaktøren der godkendte signalet til forsiden
  godkendtTid     DateTime?

  @@unique([instansId, externalId])
  @@unique([instansId, kildeUrlNorm])
  @@index([instansId, laest])
  @@index([instansId, createdAt])
  @@index([instansId, godkendtTid])
}

// ── First-Party Metrikker & Læseradfærd (Metric Engine) ────────────────────

  id               String   @id @default(cuid())

  article          Article  @relation(fields: [articleId], references: [id], onDelete: Cascade)
  instansId        String
  instans          Instance @relation(fields: [instansId], references: [id])
  visninger        Int      @default(0)
  laesninger       Int      @default(0) // Scroll > 70% eller læsetid > 30s
  totalLaesetidSek Int      @default(0)

  hourlyViews      Json?    // Seneste timer [h0, h1, ...]
  createdAt        DateTime @default(now())
  opdateretTid     DateTime @updatedAt

  @@index([instansId, score])
}

  @@index([instansId])

// ── T11/T12: modulær forside, layout, AI-forslag (additivt) ──────────────────
// Spec: docs/review/T11-T12-spec.md. Kode: lib/frontpage/**.
// Status-strenge valideres i lib/frontpage/layout-schema.ts (cross-provider: ingen enums).

/** Layout = rækkefølge af modul-instanser (JSON). Pr. instans: højst ÉN række med status "live" + valgfrit mange "kladde". */
model FrontpageLayout {
  id            String   @id @default(cuid())
  instansId     String
  instans       Instance @relation(fields: [instansId], references: [id])
  name          String
  status        String   @default("kladde") // kladde | live
  version       Int      @default(1)        // stiger ved hver gem (kladde) / publicering og rollback (live)
  schemaVersion Int      @default(1)
  modules       Json     // FrontpageLayout.modules (lib/frontpage/layout-schema.ts)
  scope         Json?    // fremtid: { dagsdel?, fra?, til? }
  createdBy     String?
  publishedBy   String?
  publishedAt   DateTime?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  versions  FrontpageLayoutVersion[]
  snapshots FrontpageSnapshot[]

  @@index([instansId, status])
}

/** Historik/rollback: én række pr. version af en layout-række. */
model FrontpageLayoutVersion {
  id         String          @id @default(cuid())
  layoutId   String
  layout     FrontpageLayout @relation(fields: [layoutId], references: [id], onDelete: Cascade)
  instansId  String
  instans    Instance        @relation(fields: [instansId], references: [id])
  version    Int
  modules    Json
  note       String?         // fx "publiceret fra kladde X", "rollback til v3"
  createdBy  String?
  createdAt  DateTime        @default(now())

  @@unique([layoutId, version])
  @@index([instansId, createdAt])
}

/** Resultat af en ranking-kørsel: forslag -> (godkendt | afvist | udløbet). Det sidst godkendte vises på forsiden. */
model FrontpageSnapshot {
  id          String           @id @default(cuid())
  instansId   String
  instans     Instance         @relation(fields: [instansId], references: [id])
  layoutId    String?
  layout      FrontpageLayout? @relation(fields: [layoutId], references: [id], onDelete: SetNull)
  status      String           @default("forslag") // forslag | godkendt | afvist | udløbet
  items       Json             // SnapshotItems (lib/frontpage/layout-schema.ts): assignments + warnings + layoutVersion
  mode        String           @default("forslag") // forslag | auto (auto = fremtid, pr. modul)
  generatedBy String           @default("deterministic") // deterministic | ai
  modelId     String?
  inputHash   String?
  godkendtAf  String?
  godkendtTid DateTime?
  afvistAf    String?
  afvistTid   DateTime?
  afvistGrund String?
  expiresAt   DateTime?
  createdAt   DateTime         @default(now())
  updatedAt   DateTime         @updatedAt

  decisions FrontpageDecision[]

  @@index([instansId, status, createdAt])
  @@index([instansId, inputHash])
}

/** Gennemsigtighedslog: hvem/hvad placerede hvilken artikel i hvilket slot og hvorfor. */
model FrontpageDecision {
  id          String            @id @default(cuid())
  snapshotId  String?           // null = handling uden snapshot (fx naturligt-sprog-kommando i editoren)
  snapshot    FrontpageSnapshot? @relation(fields: [snapshotId], references: [id], onDelete: Cascade)
  instansId   String
  instans     Instance          @relation(fields: [instansId], references: [id])
  articleId   String?
  slot        String            // "<moduleId>:<index>"
  handling    String            // placeret | afvist | fjernet | godkendt | afvist-forslag | redigeret | nl-kommando | foreslaa-layout
  kilde       String            // ai | redaktør | regel
  begrundelse String?
  konfidens   Float?
  createdAt   DateTime          @default(now())

  @@index([snapshotId])
  @@index([instansId, createdAt])
}

/** Impressions/klik pr. slot og dag (CTR pr. slot). */
model FrontpageSlotMetric {

  instans     Instance @relation(fields: [instansId], references: [id])
  moduleId    String
  slotKey     String
  articleId   String
  impressions Int      @default(0)
  clicks      Int      @default(0)
  day         String   // YYYY-MM-DD (UTC)

  @@index([instansId, day])
}

// ── AI-operatør: engangs-bekræftelser og fortryd-recepter ───────────────────
/// Én række pr. handling, som AI-operatøren har foreslået (status "pending", kræver brugerens klik) eller udført
/// ("applied", evt. med en `undo`-recept). Bevidst uden relationer (som AuditLog). `input` og `result` indeholder
/// aldrig hemmeligheder; råt bekræftelses-token gemmes aldrig, kun dets SHA-256 (`tokenHash`).
model OperatorAction {
  id        String    @id @default(cuid())
  instansId String
  userId    String
  sessionId String?
  tool      String
  risk      String    // safe-write | confirm
  status    String    // pending | applied | undone | cancelled | expired | failed
  inputHash String    // sha256 af kanonisk JSON af det validerede input
  input     Json
  tokenHash String?   @unique
  result    Json?
  undo      Json?
  expiresAt DateTime?
  usedAt    DateTime?
  createdAt DateTime  @default(now())

  @@index([instansId, userId, createdAt])
  @@index([sessionId])
}

// ── Kontrolrum: prompts, kilder og feeds som redigerbare data (additivt) ────
// Kode: lib/prompts/**, lib/engine/**. UI: /redaktion/kontrolrum. Alle rækker er instansafgrænsede.

/// Redaktionens tilretning af en prompt (nøgle valideres mod lib/prompts/registry.ts). Findes ingen aktiv række, bruges
/// standardteksten i koden. Låste sikkerhedsregler og svarformat ligger KUN i koden og kan ikke overstyres her.
model PromptTemplate {
  id          String   @id @default(cuid())
  instansId   String
  instans     Instance @relation(fields: [instansId], references: [id])
  noegle      String   // fx "opgave.headlines" | "sprog.lokal" | "rating.kilde"
  indhold     String
  aktiv       Boolean  @default(true)
  version     Int      @default(1)
  opdateretAf String?  // User.id
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  revisions   PromptRevision[]

  @@unique([instansId, noegle])
  @@index([instansId])
}

/// Uforanderlig historik: én række pr. gemt version (inkl. gendannelser), så enhver ændring kan ses og rulles tilbage.
model PromptRevision {
  id        String         @id @default(cuid())
  promptId  String
  prompt    PromptTemplate @relation(fields: [promptId], references: [id], onDelete: Cascade)
  instansId String
  instans   Instance       @relation(fields: [instansId], references: [id])
  version   Int
  indhold   String
  note      String?
  aendretAf String?        // User.id
  createdAt DateTime       @default(now())

  @@unique([promptId, version])
  @@index([instansId, createdAt])
}

/// Kilderegister: redaktionens egen vurdering af en kilde (0-100). Slås op via domæne eller navn før de indbyggede regler.
model SourceProfile {
  id        String   @id @default(cuid())
  instansId String
  instans   Instance @relation(fields: [instansId], references: [id])
  navn      String
  type      String   @default("andet") // SOURCE_TYPES (lib/ingest/schema.ts) | "nyhedsbureau" | "borger"
  domaene   String?  // normaliseret værtsnavn uden www, fx "politi.dk"
  score     Int      @default(50)      // 0-100; A >= 80, B >= 60, C >= 40, ellers D
  note      String?
  aktiv     Boolean  @default(true)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@unique([instansId, navn])
  @@index([instansId, aktiv])
  @@index([instansId, domaene])
}

/// Feed-definition: hvad agenterne skal overvåge. Hentes af agenten via GET /api/ingest/feeds (scope signals:write).
model FeedDefinition {
  id          String   @id @default(cuid())
  instansId   String
  instans     Instance @relation(fields: [instansId], references: [id])
  navn        String
  type        String   @default("rss") // rss | api | web | mail
  url         String?
  sourceType  String?  // SOURCE_TYPES
  omraadeTekst String? // geografi-hint til indtaget
  inkluder    Json     // string[] nøgleord, der skal indgå
  ekskluder   Json     // string[] nøgleord, der fravælger
  intervalMin Int      @default(30)
  aktiv       Boolean  @default(true)
  noter       String?
  kategori    String?  // fri gruppering i kildepakken: Politi, Kommune, Trafik, Medier ...
  prioritet   Int      @default(2) // 1 = P0 (drift/breaking), 2 = P1, 3 = P2
  sidstHentet DateTime? // sidste manuelle hentning i CMS
  sidsteStatus String? // ok | fejl
  sidsteAntal Int?     // nye eller ændrede signaler ved sidste hentning
  sidsteBesked String? // kort resultattekst (aldrig svarindhold)

  updatedAt   DateTime @updatedAt

  @@unique([instansId, navn])
  @@index([instansId, aktiv])
  @@index([instansId, kategori])
}

  modelId       String?
  udbyder       String?

  @@index([instansId, userId])
  @@index([udloebTid])
}
